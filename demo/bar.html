Data → markup

CssCharts.renderBar(el, {
  data: [
    { name: 'A', uv: 4000, pv: 2400 },
    { name: 'B', uv: 3000, pv: 1398 },
  ],
  series: [
    { key: 'uv', label: 'UV' },
    { key: 'pv', label: 'PV' },
  ],
  layout: 'group', // simple | group | stack | horizontal
  max: 4000,
  tooltip: true, // interestfor + popover="hint" (default)
});
